//  CLASA DE BAZA: Persoana

abstract class Persoana
{
    private string idnp;
    protected static int AnCurent = DateTime.Now.Year;

    protected string Nume { get; set; }
    protected string Prenume { get; set; }
    protected int AnNastere { get; set; }

    // Proprietate publica pentru acces la nume complet
    public string NumeComplet { get { return Nume + " " + Prenume; } }

    protected Persoana(string nume, string prenume, int anNastere)
    {
        Nume = nume;
        Prenume = prenume;
        AnNastere = anNastere;
        idnp = Guid.NewGuid().ToString("N").Substring(0, 13);
    }

    public int Varsta() { return AnCurent - AnNastere; }

    public virtual void Afisare()
    {
        Console.WriteLine("  Nume    : " + Nume + " " + Prenume);
        Console.WriteLine("  An nasc.: " + AnNastere + "  (varsta: " + Varsta() + ")");
    }

    public virtual double Salariu() { return 0; }
    public virtual double Bursa() { return 0; }
    public abstract int Tip();
    public abstract void Citire();
}

//  CLASA: Angajat

class Angajat : Persoana
{
    protected int NrOre { get; set; }
    protected int PlataOra { get; set; }

    public Angajat(string nume, string prenume, int anNastere,
                   int nrOre, int plataOra)
        : base(nume, prenume, anNastere)
    {
        NrOre = nrOre;
        PlataOra = plataOra;
    }

    public override void Afisare()
    {
        Console.WriteLine("  [ANGAJAT]");
        base.Afisare();
        Console.WriteLine("  Ore lucrate : " + NrOre);
        Console.WriteLine("  Plata/ora   : " + PlataOra + " MDL");
        Console.WriteLine("  Salariu     : " + Salariu() + " MDL");
    }

    public override void Citire()
    {
        Console.Write("  Nume: "); Nume = Console.ReadLine();
        Console.Write("  Prenume: "); Prenume = Console.ReadLine();
        Console.Write("  An nastere: "); AnNastere = int.Parse(Console.ReadLine());
        Console.Write("  Nr. ore: "); NrOre = int.Parse(Console.ReadLine());
        Console.Write("  Plata/ora: "); PlataOra = int.Parse(Console.ReadLine());
    }

    public override double Salariu() { return NrOre * PlataOra; }
    public override int Tip() { return 2; }
}

class Student : Persoana
{
    protected string Grupa { get; set; }
    protected float Media { get; set; }
    private const double BursaBaza = 1200;

    public Student(string nume, string prenume, int anNastere,
                   string grupa, float media)
        : base(nume, prenume, anNastere)
    {
        Grupa = grupa;
        Media = media;
    }

    public override void Afisare()
    {
        Console.WriteLine("  [STUDENT]");
        base.Afisare();
        Console.WriteLine("  Grupa : " + Grupa);
        Console.WriteLine("  Media : " + Media.ToString("F2"));
        Console.WriteLine("  Bursa : " + Bursa() + " MDL");
    }

    public override void Citire()
    {
        Console.Write("  Nume: "); Nume = Console.ReadLine();
        Console.Write("  Prenume: "); Prenume = Console.ReadLine();
        Console.Write("  An nastere: "); AnNastere = int.Parse(Console.ReadLine());
        Console.Write("  Grupa: "); Grupa = Console.ReadLine();
        Console.Write("  Media: "); Media = float.Parse(Console.ReadLine());
    }

    public override double Bursa()
    {
        return Media >= 7 ? BursaBaza + (Media - 7) * 100 : 0;
    }
    public override int Tip() { return 1; }
}

//  CLASA: Celula

class Celula
{
    public Persoana Persoana { get; set; }
    public Celula Urmatoarea { get; set; }

    public Celula(Persoana p)
    {
        Persoana = p;
        Urmatoarea = null;
    }

    public void Afiseaza()
    {
        Console.WriteLine(new string('-', 40));
        Persoana.Afisare();
        Console.WriteLine(new string('-', 40));
    }

    public double Consum() { return Persoana.Salariu() + Persoana.Bursa(); }
}

//  CLASA: Lista

class Lista
{
    private Celula prim;

    public Lista() { prim = null; }

    public void Inserare(Persoana p)
    {
        Celula noua = new Celula(p);
        if (prim == null) { prim = noua; return; }
        Celula curent = prim;
        while (curent.Urmatoarea != null) curent = curent.Urmatoarea;
        curent.Urmatoarea = noua;
        Console.WriteLine("  >> '" + p.GetType().Name + "' adaugat cu succes.");
    }

    public void Exclude(string numeCautat)
    {
        if (prim == null) { Console.WriteLine("  Lista este goala."); return; }

        if (ContineNume(prim.Persoana, numeCautat))
        {
            prim = prim.Urmatoarea;
            Console.WriteLine("  >> '" + numeCautat + "' eliminat.");
            return;
        }

        Celula prev = prim;
        Celula curent = prim.Urmatoarea;
        while (curent != null)
        {
            if (ContineNume(curent.Persoana, numeCautat))
            {
                prev.Urmatoarea = curent.Urmatoarea;
                Console.WriteLine("  >> '" + numeCautat + "' eliminat.");
                return;
            }
            prev = curent;
            curent = curent.Urmatoarea;
        }
        Console.WriteLine("  >> Persoana '" + numeCautat + "' nu a fost gasita.");
    }

    private static bool ContineNume(Persoana p, string n)
    {
        // IndexOf cu StringComparison e disponibil si in C# 7.3
        return p.NumeComplet.IndexOf(n, StringComparison.OrdinalIgnoreCase) >= 0;
    }

    public void Afisare()
    {
        if (prim == null) { Console.WriteLine("  Lista este goala."); return; }
        Celula curent = prim;
        int i = 1;
        while (curent != null)
        {
            Console.WriteLine("\n  [" + i++ + "]");
            curent.Afiseaza();
            curent = curent.Urmatoarea;
        }
    }

    public void AfisareDupaTip(int tip)
    {
        Celula curent = prim;
        bool gasit = false;
        while (curent != null)
        {
            if (curent.Persoana.Tip() == tip)
            {
                curent.Afiseaza();
                gasit = true;
            }
            curent = curent.Urmatoarea;
        }
        if (!gasit) Console.WriteLine("  Nu exista persoane de tipul selectat.");
    }

    public double TotalBani()
    {
        double total = 0;
        Celula curent = prim;
        while (curent != null)
        {
            total += curent.Consum();
            curent = curent.Urmatoarea;
        }
        return total;
    }

    public int Numar()
    {
        int n = 0;
        Celula c = prim;
        while (c != null) { n++; c = c.Urmatoarea; }
        return n;
    }

    public double SalariuMediu()
    {
        double sum = 0;
        int cnt = 0;
        Celula c = prim;
        while (c != null)
        {
            if (c.Persoana.Salariu() > 0) { sum += c.Persoana.Salariu(); cnt++; }
            c = c.Urmatoarea;
        }
        return cnt == 0 ? 0 : sum / cnt;
    }

    public void Cauta(string fragment)
    {
        Celula c = prim;
        bool gasit = false;
        while (c != null)
        {
            if (c.Persoana.NumeComplet.IndexOf(fragment, StringComparison.OrdinalIgnoreCase) >= 0)
            {
                c.Afiseaza();
                gasit = true;
            }
            c = c.Urmatoarea;
        }
        if (!gasit) Console.WriteLine("  Nu a fost gasit niciun rezultat.");
    }
}
