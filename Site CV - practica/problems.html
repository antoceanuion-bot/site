Problema 01

M1. 5 Se consideră mulțimea P={P1,P2,...,Pn} formată din n puncte (2≤n≤30) pe un plan euclidian. Fișierul date.in conține pe prima linie numărul n, care reprezintă numărul de puncte din plan pe liniile următoare se regăsesc coordonatele punctelor. Fiecare punct Pj este definit prin coordonatele sale xj, yj . Elaborați un program care afișează la ecran coordonatele punctelor Pa, Pb distanța dintre care este maximă. Distanța dintre punctele Pj, Pm se calculează cu ajutorul formulei: djm = √(xj − xm) 2 + (yj−ym) 2

Solutie C#

Afiseaza Codul Sursa 
using System;

class Program
{
    static void Main()
    {
        // Citirea datelor de la tastatura
        Console.Write("Introduceti numarul de puncte n (2-30): ");
        int n = int.Parse(Console.ReadLine());

        double[] x = new double[n];
        double[] y = new double[n];

        for (int i = 0; i < n; i++)
        {
            Console.Write($"Introduceti coordonatele punctului P{i + 1} (x y): ");
            string[] parts = Console.ReadLine().Trim().Split(' ');
            x[i] = double.Parse(parts[0]);
            y[i] = double.Parse(parts[1]);
        }

        // Gasirea perechii de puncte cu distanta maxima
        double distantaMax = -1;
        int indexA = 0, indexB = 1;

        for (int j = 0; j < n - 1; j++)
        {
            for (int m = j + 1; m < n; m++)
            {
                double dx = x[j] - x[m];
                double dy = y[j] - y[m];
                double distanta = Math.Sqrt(dx * dx + dy * dy);

                if (distanta > distantaMax)
                {
                    distantaMax = distanta;
                    indexA = j;
                    indexB = m;
                }
            }
        }

        // Afisarea rezultatelor
        Console.WriteLine($"Punctul Pa = P{indexA + 1}({x[indexA]}, {y[indexA]})");
        Console.WriteLine($"Punctul Pb = P{indexB + 1}({x[indexB]}, {y[indexB]})");
        Console.WriteLine($"Distanta maxima = {distantaMax:F4}");
    }
}
}

Problema 02

M2. 2 Problema rucsacului. O persoană are un rucsac cu care poate transporta o greutate maximă G. Persoana are la dispoziție n obiecte și cunoaște pentru fiecare obiect greutatea și câștigul care se obține în urma transportului său la destinație. Se cere să se precizeze ce obiecte trebuie să transporte persoana în așa fel încât câștigul să fie maxim. Persoana are posibilitatea să taie obiectele

Solutie C#

Afiseaza Codul Sursa 
using System;

class Obiect
{
    public int Index;
    public double Greutate;
    public double Castig;
    public double Raport;
}

class Program
{
    static void Main()
    {
        Console.Write("Introduceti greutatea maxima a rucsacului (G): ");
        double G = double.Parse(Console.ReadLine());

        Console.Write("Introduceti numarul de obiecte (n): ");
        int n = int.Parse(Console.ReadLine());

        Obiect[] obiecte = new Obiect[n];

        for (int i = 0; i < n; i++)
        {
            Console.Write($"Obiect {i + 1} - greutate si castig (ex: 10 60): ");
            string[] parts = Console.ReadLine().Trim().Split(' ');
            obiecte[i] = new Obiect
            {
                Index = i + 1,
                Greutate = double.Parse(parts[0]),
                Castig = double.Parse(parts[1]),
                Raport = double.Parse(parts[1]) / double.Parse(parts[0])
            };
        }

        // Greedy
        Array.Sort(obiecte, (a, b) => b.Raport.CompareTo(a.Raport));

        double castigTotal = 0;
        double greutateRamas = G;

        Console.WriteLine("\n--- Obiecte selectate ---");

        for (int i = 0; i < n; i++)
        {
            if (greutateRamas <= 0)
                break;

            if (obiecte[i].Greutate <= greutateRamas)
            {
                // Luam obiectul intreg
                Console.WriteLine($"Obiect {obiecte[i].Index}: luat complet " +
                                  $"(greutate={obiecte[i].Greutate}, castig={obiecte[i].Castig})");
                castigTotal += obiecte[i].Castig;
                greutateRamas -= obiecte[i].Greutate;
            }
            else
            {
                // Luam doar o fractiune din obiect
                double fractiune = greutateRamas / obiecte[i].Greutate;
                double castigPartial = fractiune * obiecte[i].Castig;
                Console.WriteLine($"Obiect {obiecte[i].Index}: luat {fractiune * 100:F1}% " +
                                  $"(greutate={greutateRamas:F2}, castig={castigPartial:F2})");
                castigTotal += castigPartial;
                greutateRamas = 0;
            }
        }

        Console.WriteLine($"\nCastig total maxim = {castigTotal:F2}");
        Console.WriteLine($"Greutate utilizata  = {G - greutateRamas:F2} / {G}");
    }
}
}

Problema 03

M3. 2 Elaborați un program care afișează la ecran toate modurile de a descompune un număr natural în sumă de k numere naturale distincte. Exemplu: pentru n=9 și k=3 soluțiile sunt: 1+2+6,2+3+4,1+3+5 3.

Solutie C#

Afiseaza Codul Sursa 
using System;

class Program
{
    static int n, k;
    static int[] sol;
    static int nrSolutii = 0;

    static void Backtrack(int nivel, int sumaRamasa, int minVal)
    {
        // suma e 0 => solutie gasita
        if (nivel == k)
        {
            if (sumaRamasa == 0)
            {
                nrSolutii++;
                for (int i = 0; i < k; i++)
                {
                    if (i > 0) Console.Write("+");
                    Console.Write(sol[i]);
                }
                Console.WriteLine();
            }
            return;
        }


        int ramase = k - nivel - 1; 

        for (int val = minVal; val <= sumaRamasa - (ramase * (2 * minVal + ramase + 1) / 2); val++)
        {
            sol[nivel] = val;
            Backtrack(nivel + 1, sumaRamasa - val, val + 1);
        }
    }

    static void Main()
    {
        Console.Write("Introduceti n: ");
        n = int.Parse(Console.ReadLine());

        Console.Write("Introduceti k: ");
        k = int.Parse(Console.ReadLine());

        sol = new int[k];

        Console.WriteLine($"\nDescompunerile lui {n} in suma de {k} numere naturale distincte:");
        Console.WriteLine("---------------------------------------------------");

        Backtrack(0, n, 1);

        Console.WriteLine("---------------------------------------------------");
        Console.WriteLine($"Total solutii: {nrSolutii}");
    }
}

Problema 04

M3. 38 Generarea combinărilor. Să se genereze combinări din n elemente luate câte m. Numărul combinărilor a n elemente luate câte m este calculată cu formula: Cn m = n! m!(n−m)!

Solutie C#

Afiseaza Codul Sursa 
using System;

class Program
{
    static int n, m;
    static int[] sol;
    static int nrCombinari = 0;

    static void Backtrack(int nivel, int minVal)
    {
        if (nivel == m)
        {
            nrCombinari++;
            Console.Write("{ ");
            for (int i = 0; i < m; i++)
            {
                if (i > 0) Console.Write(", ");
                Console.Write(sol[i]);
            }
            Console.WriteLine(" }");
            return;
        }

        for (int val = minVal; val <= n - (m - nivel - 1); val++)
        {
            sol[nivel] = val;
            Backtrack(nivel + 1, val + 1);
        }
    }

    static long Factorial(int x)
    {
        long rez = 1;
        for (int i = 2; i <= x; i++)
            rez *= i;
        return rez;
    }

    static void Main()
    {
        Console.Write("Introduceti n: ");
        n = int.Parse(Console.ReadLine());

        Console.Write("Introduceti m: ");
        m = int.Parse(Console.ReadLine());

        if (m > n)
        {
            Console.WriteLine("Eroare: m nu poate fi mai mare decat n!");
            return;
        }

        sol = new int[m];

        long C = Factorial(n) / (Factorial(m) * Factorial(n - m));

        Console.WriteLine($"\nCombinarile din {n} elemente luate cate {m} (C({n},{m}) = {C}):");
        Console.WriteLine("---------------------------------------------------");

        Backtrack(0, 1);

        Console.WriteLine("---------------------------------------------------");
        Console.WriteLine($"Total combinari generate: {nrCombinari}");
    }
}

Problema 05

M4. 2 Să se genereze și să se numere toate permutările mulțimii {1,2,3,...,n} care încep cu valoarea 1.

using System;
using System.Collections.Generic;
class PermutariIncepCu1
{
    static int n;           
    static int[] perm;     
    static bool[] folosit;  
    static int contor = 0;  

    static void Main()
    {
        Console.Write("Introdu n (dimensiunea multimii {1, 2, ..., n}): ");
        n = int.Parse(Console.ReadLine());

        perm = new int[n + 1]; 
        folosit = new bool[n + 1];

    
        perm[1] = 1;
        folosit[1] = true;

        Console.WriteLine($"\nPermutarile lui {{1..{n}}} care incep cu 1:\n");

        //  backtracking
        Genereaza(2);

        Console.WriteLine($"\nTotal permutari: {contor}  (= {n - 1}! = {Factorial(n - 1)})");
    }

    static void Genereaza(int pos)
    {
        if (pos == n + 1)
        {
            
            contor++;
            Console.Write($"{contor,4}. (");
            for (int i = 1; i <= n; i++)
                Console.Write(i < n ? $"{perm[i]}, " : $"{perm[i]}");
            Console.WriteLine(")");
            return;
        }

      
        for (int val = 1; val <= n; val++)
        {
            if (!folosit[val])
            {
                perm[pos] = val;
                folosit[val] = true;

                Genereaza(pos + 1);

                // Backtrack
                folosit[val] = false;
            }
        }
    }

    /// Calculează n! iterativ.
    static long Factorial(int k)
    {
        long rez = 1;
        for (int i = 2; i <= k; i++)
            rez *= i;
        return rez;
    }
}

Problema 06

M5. 24 Calculați subsecvența de sumă maximă pentru un vector A de n numere reale folosind o abordare de tip desparte și stăpânește. Subsecvență a unui vector A[1..n] se definește ca o secvență continuă de elemente din A. De exemplu, A[i..j](pentru orice 1<=i<=j<=n) reprezintă o subsecvență a lui A. Exemplu: Pentru A = {−2, 1, −3, 4, −1, 2, 1, −5, 4}, subsecvența de sumă maximă este {4, −1, 2, 1} de sumă 6.

Solutie C#

Afiseaza Codul Sursa 
using System;
class SubsecventaSumaMaxima
{
    //Structura pentru a reține rezultatul complet 
    struct Rezultat
    {
        public double Suma;  
        public int Inceput; 
        public int Sfarsit; 

        public Rezultat(double suma, int inc, int sf)
        { Suma = suma; Inceput = inc; Sfarsit = sf; }

        public override string ToString() =>
            $"suma = {Suma},  indici [{Inceput}..{Sfarsit}]";
    }

    static double[] A; 


    //  CAZUL 3 — subsecvența care traversează mijlocul

    static Rezultat SubsecventaCuMijloc(int st, int mid, int dr)
    {
        double sumStanga = double.NegativeInfinity;
        double suma = 0;
        int idxStanga = mid;
        for (int i = mid; i >= st; i--)
        {
            suma += A[i];
            if (suma > sumStanga)
            {
                sumStanga = suma;
                idxStanga = i;
            }
        }

        double sumDreapta = double.NegativeInfinity;
        suma = 0;
        int idxDreapta = mid + 1;
        for (int j = mid + 1; j <= dr; j++)
        {
            suma += A[j];
            if (suma > sumDreapta)
            {
                sumDreapta = suma;
                idxDreapta = j;
            }
        }

        return new Rezultat(sumStanga + sumDreapta, idxStanga, idxDreapta);
    }

    //  DIVIDE AND CONQUER PRINCIPAL

    static Rezultat MaxSubarray(int st, int dr)
    {
        if (st == dr)
            return new Rezultat(A[st], st, st);

        int mid = (st + dr) / 2;

        Rezultat rezStanga = MaxSubarray(st, mid);       
        Rezultat rezDreapta = MaxSubarray(mid + 1, dr);  
        Rezultat rezMijloc = SubsecventaCuMijloc(st, mid, dr); 

        if (rezStanga.Suma >= rezDreapta.Suma && rezStanga.Suma >= rezMijloc.Suma)
            return rezStanga;
        if (rezDreapta.Suma >= rezStanga.Suma && rezDreapta.Suma >= rezMijloc.Suma)
            return rezDreapta;
        return rezMijloc;
    }

    //  MAIN
    static void Main()
    {
        // Exemplul din problemă
        Console.WriteLine("=== Exemplu din problema ===");
        double[] exemplu = { 0, -2, 1, -3, 4, -1, 2, 1, -5, 4 }; 
        int n = exemplu.Length - 1; 

        A = exemplu;
        AfiseazaVector(n);

        Rezultat rez = MaxSubarray(1, n);
        AfiseazaRezultat(rez, n);

        // ── Exemplu cu toate negative 
        Console.WriteLine("\n=== Toate elementele negative ===");
        A = new double[] { 0, -5, -1, -3, -2, -4 };
        n = A.Length - 1;
        AfiseazaVector(n);
        rez = MaxSubarray(1, n);
        AfiseazaRezultat(rez, n);

        // ── Exemplu cu un singur element 
        Console.WriteLine("\n=== Un singur element ===");
        A = new double[] { 0, 7.5 };
        n = 1;
        AfiseazaVector(n);
        rez = MaxSubarray(1, n);
        AfiseazaRezultat(rez, n);

        // Citire de la tastatură 
        Console.WriteLine("\n=== Introdu propriul vector ===");
        Console.Write("n = ");
        n = int.Parse(Console.ReadLine());
        A = new double[n + 1];
        Console.WriteLine("Introdu elementele (spatiu-separate):");
        string[] tokens = Console.ReadLine().Split(' ');
        for (int i = 1; i <= n; i++)
            A[i] = double.Parse(tokens[i - 1]);

        AfiseazaVector(n);
        rez = MaxSubarray(1, n);
        AfiseazaRezultat(rez, n);
    }

    //  UTILITARE DE AFIȘARE
    static void AfiseazaVector(int n)
    {
        Console.Write("A = { ");
        for (int i = 1; i <= n; i++)
            Console.Write(i < n ? $"{A[i]}, " : $"{A[i]}");
        Console.WriteLine(" }");
    }

    static void AfiseazaRezultat(Rezultat rez, int n)
    {
        Console.WriteLine($"Subsecventa de suma maxima:");
        Console.Write("  Elemente: { ");
        for (int i = rez.Inceput; i <= rez.Sfarsit; i++)
            Console.Write(i < rez.Sfarsit ? $"{A[i]}, " : $"{A[i]}");
        Console.WriteLine(" }");
        Console.WriteLine($"  Suma    : {rez.Suma}");
        Console.WriteLine($"  Pozitii : A[{rez.Inceput}..{rez.Sfarsit}]");
    }
}

Problema 07

Introdu M6. 2 Problema diligenței. Un comis-voiajor are de făcut o călătorie cu diligența între două orașe din vestul sălbatic. Călătoria cu diligența se desfășoară în n etape. În fiecare etapă diligența merge, fără întrerupere, între două orașe. După o etapă diligența este schimbată, iar comis-voiajorul decide în care oraș va merge în etapa următoare. Se știe că în fiecare etapă, cu excepția ultimei etape, se poate ajunge în unul din cele k orașe. Se cunoaște orașul de start, S și cel final, F (în ultima etapă se ajunge în orașul F). Pentru orice etapă se cunosc toate distanțele între orașele care sunt puncte de pornire și cele care sunt puncte de destinație. class ProblemaDiligentei
{

    static void RuleazaExempluClassic()
    {

        int[][][] dist = new int[5][][];

        dist[1] = new int[1][];
        dist[1][0] = new int[] { 2, 4, 3 };

        dist[2] = new int[][] {
            new int[] { 7, 4, 6 },

            new int[] { 4, 3, 5 }
        };

        dist[3] = new int[][] {
            new int[] { 1, 4, 7 },
            new int[] { 6, 3, 3 },
            new int[] { 5, 3, 2 }
        };

        // Etapa 4
        dist[4] = new int[][] {
            new int[] { 5 },
            new int[] { 2 },

        };

        Console.WriteLine("  EXEMPLU CLASIC");

        int[] nrSurse = { 0, 1, k, k, k };
        int[] nrDest = { 0, k, k, k, 1 };

        for (int t = 1; t <= n; t++)
        {
            Console.WriteLine($"  Etapa {t} ({etapeDesc[t]}):");

            Console.Write("         ");
            if (t < n)
                for (int j = 0; j < k; j++) Console.Write($"  Oraș{j + 1}");
            else
                Console.Write("     F");
            Console.WriteLine();

            for (int i = 0; i < nrSurse[t]; i++)
            {
                string src = (t == 1) ? "  S    " : $"  Oraș{i + 1} ";
                Console.Write(src);
                for (int j = 0; j < nrDest[t]; j++)
                    Console.Write($"  {dist[t][i][j],4}");
                Console.WriteLine();
            }
            Console.WriteLine();
        }

        RezolvaSiAfiseaza(n, k, dist, nrSurse, nrDest);
    }

    //  CITIRE INTERACTIVĂ
    static void RuleazaInteractiv()
    {
        Console.Write("Număr de etape n = ");
        int n = int.Parse(Console.ReadLine());
        Console.Write("Număr de orașe intermediare k = ");
        int k = int.Parse(Console.ReadLine());

        int[] nrSurse = new int[n + 1];
        int[] nrDest = new int[n + 1];
        for (int t = 1; t <= n; t++)
        {
            nrSurse[t] = (t == 1) ? 1 : k;
            nrDest[t] = (t == n) ? 1 : k;
        }

        int[][][] dist = new int[n + 1][][];
        for (int t = 1; t <= n; t++)
        {
            dist[t] = new int[nrSurse[t]][];
            string dstLabel = (t == n) ? "F" : $"{k} orașe";
            string srcLabel = (t == 1) ? "S" : $"{k} orașe";
            Console.WriteLine($"\nEtapa {t}: {srcLabel} → {dstLabel}");

            for (int i = 0; i < nrSurse[t]; i++)
            {
                dist[t][i] = new int[nrDest[t]];
                string src = (t == 1) ? "S" : $"oraș {i + 1}";
                for (int j = 0; j < nrDest[t]; j++)
                {
                    string dst = (t == n) ? "F" : $"oraș {j + 1}";
                    Console.Write($"  dist({src} → {dst}) = ");
                    string inp = Console.ReadLine().Trim();
                    dist[t][i][j] = (inp == "-" || inp.ToLower() == "inf") ? INF
                                  : int.Parse(inp);
                }
            }
        }

        RezolvaSiAfiseaza(n, k, dist, nrSurse, nrDest);
    }

    //  NUCLEUL DP
    static void RezolvaSiAfiseaza(int n, int k,
                                   int[][][] dist,
                                   int[] nrSurse, int[] nrDest)
    {
        int[][] dp = new int[n + 2][];
        int[][] next = new int[n + 1][];

        // Condiție de frontieră: în F costul e 0
        dp[n + 1] = new int[] { 0 };

        // Umplere înapoi
        for (int t = n; t >= 1; t--)
        {
            dp[t] = new int[nrSurse[t]];
            next[t] = new int[nrSurse[t]];

            for (int i = 0; i < nrSurse[t]; i++)
            {
                dp[t][i] = INF;
                next[t][i] = -1;

                for (int j = 0; j < nrDest[t]; j++)
                {
                    if (dist[t][i][j] >= INF) continue;
                    int cost = dist[t][i][j] + dp[t + 1][j];
                    if (cost < dp[t][i])
                    {
                        dp[t][i] = cost;
                        next[t][i] = j;
                    }
                }
            }
        }

        // Tabel DP 
        Console.WriteLine("\n┌─────────────────────────────────────────────────┐");
        Console.WriteLine("│           TABELUL DP (backward)                 │");
        Console.WriteLine("├──────────┬──────────┬────────────┬──────────────┤");
        Console.WriteLine("│  Etapa   │  Oraș    │  dp (min)  │  Merge în    │");
        Console.WriteLine("├──────────┼──────────┼────────────┼──────────────┤");

        for (int t = 1; t <= n; t++)
        {
            for (int i = 0; i < nrSurse[t]; i++)
            {
                string orasName = (t == 1) ? "S" : $"Oraș {i + 1}";
                string dpVal = dp[t][i] >= INF ? "∞" : dp[t][i].ToString();
                string destName = next[t][i] == -1 ? "-"
                                 : (t == n ? "F" : $"Oraș {next[t][i] + 1}");
                Console.WriteLine($"│  {t,-8}│  {orasName,-8}│  {dpVal,-10}│  {destName,-12}│");
            }
            if (t < n)
                Console.WriteLine("├──────────┼──────────┼────────────┼──────────────┤");
        }
        Console.WriteLine("└──────────┴──────────┴────────────┴──────────────┘");

        //  Rezultat 
        Console.WriteLine();
        if (dp[1][0] >= INF)
        {
            Console.WriteLine("  ✗ Nu există drum de la S la F!");
            return;
        }

        Console.WriteLine($"  Distanța minimă totală: {dp[1][0]}");
        Console.Write("  Drumul optim: S");

        int cur = 0;
        int distAcum = 0;
        for (int t = 1; t <= n; t++)
        {
            int ales = next[t][cur];
            int dPas = dist[t][cur][ales];
            distAcum += dPas;
            string dest = (t == n) ? "F" : $"Oraș {ales + 1}";
            Console.Write($" →[{dPas}]→ {dest}");
            cur = ales;
        }
        Console.WriteLine($"\n  Verificare sumă: {distAcum}");
    }

    //  MAIN
    static void Main()
    {

Date de ieșire: Fișierul de ieșire date.out va conține pe prima linie un singur număr natural reprezentând suma maximă posibilă. Nodurile 2 și 3 au ca nod tată pe 1, iar nodurile 4 și 5 au ca tată pe 3. Nodul 1 are asociată valoarea 3, nodurile 2 și 3 au asociată valoarea 4, nodul 4 are asociată valoarea 5, iar nodul 5 are valoarea 3 asociată. Afiseaza Codul Sursa 
using System;
using System.Collections.Generic;
using System.IO;
class SumaMaximaArbore
{
    static void Main()
    {
        // Citire date
        int[] tata;
        int[] val;
        int n;

        if (File.Exists("date.in"))
        {
            string[] lines = File.ReadAllLines("date.in");
            n = int.Parse(lines[0].Trim());
            tata = Array.ConvertAll(lines[1].Trim().Split(' '), int.Parse);
            val = Array.ConvertAll(lines[2].Trim().Split(' '), int.Parse);
        }
        else
        {
            // Exemplul din problemă dacă fișierul nu există
            Console.WriteLine("[date.in negăsit — folosim exemplul din problemă]\n");
            n = 5;
            tata = new int[] { 0, 1, 1, 3, 3 };
            val = new int[] { 3, 4, 4, 5, 3 };
        }

        List[] fii = new List[n + 1];
        for (int i = 1; i <= n; i++)
            fii[i] = new List();

        int radacina = -1;
        for (int i = 1; i <= n; i++)
        {
            int t = tata[i - 1];
            if (t == 0)
                radacina = i;
            else
                fii[t].Add(i);
        }

        int[,] dp = new int[n + 1, 2];

        int[] ordine = new int[n];
        int head = 0, tail = 0;
        ordine[tail++] = radacina;
        while (head < tail)
        {
            int nod = ordine[head++];
            foreach (int fiu in fii[nod])
                ordine[tail++] = fiu;
        }

        for (int idx = n - 1; idx >= 0; idx--)
        {
            int i = ordine[idx];
            dp[i, 1] = val[i - 1]; 

            if (fii[i].Count == 0)
            {
                dp[i, 0] = 0;
            }
            else
            {
                foreach (int fiu in fii[i])
                {
                    dp[i, 1] += dp[fiu, 0];
                    dp[i, 0] += Math.Max(dp[fiu, 0], dp[fiu, 1]);
                }
            }
        }

        int sumaMaxima = Math.Max(dp[radacina, 0], dp[radacina, 1]);

        // econstrucție noduri alese
        List noduriAlese = new List();
        // Parcurgem BFS și decidem pentru fiecare nod dacă e ales sau nu
        bool[] ales = new bool[n + 1];
        // Folosim o coadă: (nod, esteAles)
        Queue<(int nod, bool parinteAles)> coada = new Queue<(int, bool)>();
        bool startAles = dp[radacina, 1] > dp[radacina, 0];
        coada.Enqueue((radacina, startAles));
        if (startAles) ales[radacina] = true;

        while (coada.Count > 0)
        {
            var (nod, nodAles) = coada.Dequeue();
            if (nodAles) noduriAlese.Add(nod);

            foreach (int fiu in fii[nod])
            {
                if (nodAles)
                {
                    coada.Enqueue((fiu, false));
                }
                else
                {
                    bool fiuAles = dp[fiu, 1] >= dp[fiu, 0];
                    coada.Enqueue((fiu, fiuAles));
                }
            }
        }
        noduriAlese.Sort();

        // Afișare rezultate 
        Console.WriteLine("══════════════════════════════════════════");
        Console.WriteLine("  M6.25 — Suma maximă pe arbore (DP)");
        Console.WriteLine("══════════════════════════════════════════");
        Console.WriteLine($"\n  n = {n}  |  Rădăcină = {radacina}");

        Console.Write("\n  Structura arborelui (tată[i]):");
        for (int i = 1; i <= n; i++)
            Console.Write($"  {i}→{tata[i - 1]}");

        Console.Write("\n  Valorile nodurilor (v[i]):  ");
        for (int i = 1; i <= n; i++)
            Console.Write($"  {i}:{val[i - 1]}");

        Console.WriteLine("\n\n  Tabel DP:");
        Console.WriteLine("  ┌────────┬──────────┬──────────┬──────────┐");
        Console.WriteLine("  │  Nod   │  v[i]    │ dp[i][0] │ dp[i][1] │");
        Console.WriteLine("  ├────────┼──────────┼──────────┼──────────┤");
        for (int i = 1; i <= n; i++)
        {
            string marca = noduriAlese.Contains(i) ? " ✓" : "  ";
            Console.WriteLine($"  │  {i,3}{marca} │  {val[i - 1],6}  │  {dp[i, 0],6}  │  {dp[i, 1],6}  │");
        }

        Console.Write(string.Join(", ", noduriAlese));
        Console.WriteLine("}");

        Console.Write("  Valori:       {");
        Console.Write(string.Join(", ", noduriAlese.ConvertAll(x => val[x - 1])));
        Console.WriteLine("}");

        Console.WriteLine($"\n  Suma maximă = {sumaMaxima}");

        Console.WriteLine("\n  [Rezultatul a fost scris în date.out]");
    }

    {

abstract class ComponentaPC
{
    public string CodProdus { get; set; }
    public double Pret { get; set; }
    public string Marca { get; set; }
    public int AnFabricatie { get; set; }
    public int Garantie { get; set; }
    public string Producator { get; set; }
    public string TaraProvenienta { get; set; }
    public int Stoc { get; set; }

    public abstract string Categorie { get; }

    protected ComponentaPC() { }

    protected ComponentaPC(string cod, double pret, string marca, int an,
                            int garantie, string producator, string tara, int stoc)
    {
        CodProdus = cod;
        Pret = pret;
        Marca = marca;
        AnFabricatie = an;
        Garantie = garantie;
        Producator = producator;
        TaraProvenienta = tara;
        Stoc = stoc;
    }

    public virtual string ToCsv()
        => Categorie + ";" + CodProdus + ";" + Pret + ";" + Marca + ";" +
           AnFabricatie + ";" + Garantie + ";" + Producator + ";" + TaraProvenienta + ";" + Stoc;

    public virtual void CitesteCampuriExtra(string[] parts) { }

    public override string ToString()
        => string.Format("  [{0}] {1,-16} | {2,-14} | Pret: {3,8:F2} lei | An: {4} | Gar: {5} luni | {6,-12} | {7,-10} | Stoc: {8}",
            CodProdus, Marca, Categorie, Pret, AnFabricatie, Garantie, Producator, TaraProvenienta, Stoc);
}

class Depozit where T : ComponentaPC
{
    private List _produse = new List();

    public IReadOnlyList Produse { get { return _produse.AsReadOnly(); } }

    public void Adauga(T produs) { _produse.Add(produs); }

    public bool Sterge(string cod)
    {
        T p = _produse.FirstOrDefault(x => x.CodProdus == cod);
        if (p == null) return false;
        _produse.Remove(p);
        return true;
    }

    public T Cauta(string cod)
        => _produse.FirstOrDefault(x => x.CodProdus == cod);

    public bool Vinde(string cod)
    {
        T p = Cauta(cod);
        if (p == null || p.Stoc <= 0) return false;
        p.Stoc--;
        return true;
    }

    public bool IncarcaStoc(string cod, int cantitate)
    {
        T p = Cauta(cod);
        if (p == null) return false;
        p.Stoc += cantitate;
        return true;
    }

    public double TotalValoare()
        => _produse.Sum(p => p.Stoc * p.Pret);

    public List CautaDupaMarca(string marca)
        => _produse.Where(p => p.Marca.IndexOf(marca, StringComparison.OrdinalIgnoreCase) >= 0).ToList();

    public void Afiseaza()
    {
        if (!_produse.Any()) { Console.WriteLine("  (depozit gol)"); return; }
        foreach (T p in _produse.OrderBy(x => x.Marca))
            Console.WriteLine(p);
    }
}

    {
        // 1. Oricare doua numere sa coincida exact in m pozitii

        for (int a = 0; a < 5; a++)
        {
            for (int b = a + 1; b < 5; b++)
            {
                // Numaram coincidentele in primele 'nivel' pozitii
                int coincidente = 0;
                for (int pos = 0; pos < nivel; pos++)
                    if (sol[pos, a] == sol[pos, b])
                        coincidente++;

                // Daca am completat toate cifrele: exact m coincidente
                if (nivel == n && coincidente != m)
                    return false;


                int ramase = n - nivel;
                if (coincidente + ramase < m) return false;  
                if (coincidente > m) return false;  
            }
        }

        // 2. Sa nu existe o pozitie in care aceeasi cifra apare in TOATE 5 numerele
        for (int pos = 0; pos < nivel; pos++)
        {
            bool totiEgali = true;
            for (int nr = 1; nr < 5; nr++)
                if (sol[pos, nr] != sol[pos, 0])
                { totiEgali = false; break; }

            if (totiEgali) return false;
        }

        return true;
    }

    // Backtracking: completam cifra 'nivel' pentru toate cele 5 numere
    static void Back(int nivel)
    {
        if (gasit) return;

        if (nivel == n)
        {
            if (Verifica(nivel))
                gasit = true;
            return;
        }

        for (int mask = 0; mask < 32 && !gasit; mask++)
        {
            for (int nr = 0; nr < 5; nr++)
                sol[nivel, nr] = ((mask >> nr) & 1) == 0 ? 1 : 2;

            if (Verifica(nivel + 1))
                Back(nivel + 1);
        }
    }

    static void Main()
    {
        Console.Write("Introduceti n (numarul de cifre): ");
        n = int.Parse(Console.ReadLine().Trim());

        Console.Write("Introduceti m (coincidente exacte intre oricare 2 numere): ");
        m = int.Parse(Console.ReadLine().Trim());

        // Validari
        if (m < 0 || m > n)
        {
            Console.WriteLine("Eroare: m trebuie sa fie intre 0 si n!");
            return;
        }

        sol = new int[n, 5];

        Console.WriteLine("\nSe cauta solutia...");
        Back(0);

        if (gasit)
        {
            Console.WriteLine("\nSolutie gasita:\n");
            Console.WriteLine("  Pos:  " + string.Join("  ", System.Linq.Enumerable.Range(1, n)));
            Console.WriteLine("  " + new string('-', 4 + n * 3));

            for (int nr = 0; nr < 5; nr++)
            {
                Console.Write("  Nr" + (nr + 1) + ": ");
                for (int pos = 0; pos < n; pos++)
                    Console.Write(sol[pos, nr] + "  ");
                Console.WriteLine();
            }

            // Verificare afisata
            Console.WriteLine("\n  Verificare coincidente intre perechi:");
            for (int a = 0; a < 5; a++)
            {
                for (int b = a + 1; b < 5; b++)
                {
                    int coin = 0;
                    for (int pos = 0; pos < n; pos++)
                        if (sol[pos, a] == sol[pos, b]) coin++;
                    Console.WriteLine("  Nr" + (a + 1) + " si Nr" + (b + 1) + ": " + coin + " coincidente");
                }
            }

            Console.WriteLine("\n  Verificare pozitii cu aceeasi cifra in toate 5:");
            bool existaPozitie = false;
            for (int pos = 0; pos < n; pos++)
            {
                bool totiEgali = true;
                for (int nr = 1; nr < 5; nr++)
                    if (sol[pos, nr] != sol[pos, 0]) { totiEgali = false; break; }
                if (totiEgali) { existaPozitie = true; Console.WriteLine("  Pozitia " + (pos + 1) + " are cifra identica!"); }
            }
            if (!existaPozitie)
                Console.WriteLine("  Nicio pozitie cu aceeasi cifra in toate 5. OK!");
        }
        else
        {
            Console.WriteLine("\nNu exista solutie pentru n=" + n + " si m=" + m + "!");
        }
    }
